import { createAggregate, createEntity, createMixin } from '@redemeine/aggregate';
import type { AggregateStateView, ShorthandCommandWithMeta } from '@redemeine/aggregate';

type State = { count: number };
type Payload = { name: string };
const root = createAggregate('shorthand', { count: 0 }).commands(() => ({
  zero: () => [],
  stateOnly(state) { state.count.toFixed(); return []; },
  required(state, payload: Payload) {
    // @ts-expect-error Command state remains readonly.
    state.count++;
    return { type: 'required', payload };
  },
  optional: (_state, payload?: Payload) => ({ type: 'optional', payload }),
  primitive: (_state, payload?: string) => ({ type: 'primitive', payload }),
  number: (_state, payload: number) => ({ type: 'number', payload }),
  requiredUndefined: (_state, payload: Payload | undefined) => ({ type: 'union', payload }),
  literal: (_state, payload: 'yes') => ({ type: 'literal', payload }),
  fixed: (_state, ...payload: [Payload]) => ({ type: 'fixed', payload: payload[0] }),
  fixedOptional: (_state, ...payload: [Payload?]) => ({ type: 'fixedOptional', payload: payload[0] }),
  fixedZero: (_state, ..._payload: []) => [],
  meta: { handler: (_state: AggregateStateView<State>, payload?: Payload) => ({ type: 'meta', payload }), meta: { description: 'optional' } },
  packed: {
    pack: (name: string, amount: number) => ({ name, amount }),
    handler: (_state: AggregateStateView<State>, payload: { name: string; amount: number }) => ({ type: 'packed', payload })
  }
}));
export const host = root.commands((_emit, ctx) => ({

    const noPayload = ctx.commands.zero(undefined);
    const nothing: void = noPayload.payload;
    return { type: 'forward', payload: [payload, absent.payload, nothing] };
  }
})).build();

const two = { go: (_state: AggregateStateView<State>, name: string, amount: number) => ({ type: 'bad', payload: amount.toFixed() }) };
const optionalSecond = { go(_state: AggregateStateView<State>, name: string, amount?: number) { return { type: 'bad', payload: amount }; } };
const rest = { go: (_state: AggregateStateView<State>, ...names: string[]) => ({ type: 'bad', payload: names }) };
const metaTwo = { go: { handler: two.go } };
const satisfiesTwo = metaTwo satisfies Record<string, ShorthandCommandWithMeta<State>>;
const widened: Record<string, ShorthandCommandWithMeta<State>> = metaTwo;
const genericRest = { go: <Args extends string[]>(_state: AggregateStateView<State>, ...names: Args) => ({ type: 'bad', payload: names }) };
// @ts-expect-error Two payload slots require explicit pack.
root.commands(() => two);
// @ts-expect-error An optional second slot can still accept two payload arguments.
root.commands(() => optionalSecond);
// @ts-expect-error An unbounded rest cannot prove a single payload slot.
root.commands(() => rest);
// @ts-expect-error Generic rest arguments do not prove a bounded payload slot.
root.commands(() => genericRest);
// @ts-expect-error Metadata does not bypass shorthand validation.
root.commands(() => metaTwo);
// @ts-expect-error Satisfies preserves the invalid signature.
root.commands(() => satisfiesTwo);
// @ts-expect-error Widening to a public declaration type cannot prove safe arity.
root.commands(() => widened);
