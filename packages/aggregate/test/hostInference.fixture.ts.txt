import { createAggregate, createEntity, createMixin } from '@redemeine/aggregate';

type State = { id: string; rows: { id: string; label: string }[] };
const initial: State = { id: 'one', rows: [] };
export const entity = createEntity<State, 'original'>('original').commands(() => ({
  go: (_state, payload: { mode: 'fast'; count: number }) => [],
  stop: () => [],
  maybe: { pack: (label?: 'ok') => ({ label }), handler: () => [] },
})).build();
export const named = createAggregate('host', {}).entities({ item: entity }).build();
export const direct = createAggregate('host', {}).entities(undefined, entity).build();
export const listed = createAggregate('host', {}).entityList('item', entity).build();
export const mixed = createAggregate('host', {}).entities({ alias: entity }, entity).build();
const payload: { mode: 'fast'; count: number } = named.commandCreators.itemGo({ mode: 'fast', count: 1 }).payload;
const same: typeof listed.commandCreators.itemGo = named.commandCreators.itemGo;
const variadic: typeof direct.commandCreators.originalGo = mixed.commandCreators.originalGo;
mixed.commandCreators.aliasGo(payload);
named.commandCreators.itemStop();
named.commandCreators.itemMaybe();
named.commandCreators.itemMaybe('ok');
// @ts-expect-error Required payload cannot be omitted.
named.commandCreators.itemGo();
// @ts-expect-error Literal payload types are preserved.
named.commandCreators.itemGo({ mode: 'slow', count: 1 });
// @ts-expect-error Optional arguments retain their type.
named.commandCreators.itemMaybe(123);
// @ts-expect-error Mount alias replaces the entity's original name.
named.commandCreators.originalGo(payload);
// @ts-expect-error No invented index signature.
named.commandCreators.missing();

export const selectedEntity = createEntity<State, 'selected'>('selected').selectors({
  by: (state, id: string) => state.rows.find(row => row.id === id),
  optional: (state, prefix?: string) => prefix ? state.id + prefix : state.id,
  rest: (state, ...ids: string[]) => state.rows.filter(row => ids.includes(row.id)),
  literal: () => 'ready' as const,
  readonly: state => {
    // @ts-expect-error Contextual state remains deeply readonly.
    state.rows.push({ id: 'x', label: 'x' });
    return state.rows;
  },
}).commands((_emit, ctx) => ({
  check: state => {
    const row: { readonly id: string; readonly label: string } | undefined = ctx.selectors.by(state, 'one');
    // @ts-expect-error Required selector argument remains required in command context.
    ctx.selectors.by(state);
    return [];
  },
})).build();
export const mixin = createMixin<State>().selectors({
  by: (state, id: string) => state.rows.find(row => row.id === id),
  optional: (state, id?: string) => id ?? state.id,
  rest: (state, ...ids: string[]) => ids.includes(state.id),
}).build();
const row: { readonly id: string; readonly label: string } | undefined = selectedEntity.selectors.by(initial, 'one');
const literal: 'ready' = selectedEntity.selectors.literal();
const text: string = mixin.selectors.optional(initial);
const flag: boolean = mixin.selectors.rest(initial, 'a', 'b');
selectedEntity.selectors.optional(initial, 'prefix');
selectedEntity.selectors.rest(initial, 'a', 'b');
// @ts-expect-error Required entity selector argument cannot be omitted.
selectedEntity.selectors.by(initial);
// @ts-expect-error Required mixin selector argument cannot be omitted.
mixin.selectors.by(initial);
// @ts-expect-error Selector arguments do not widen to unknown or any.
mixin.selectors.by(initial, 42);
// @ts-expect-error Optional selector arguments retain their type.
selectedEntity.selectors.optional(initial, 42);
// @ts-expect-error Rest selector arguments retain their type.
mixin.selectors.rest(initial, 42);
// @ts-expect-error Selector return views remain readonly.
selectedEntity.selectors.readonly(initial).push({ id: 'x', label: 'x' });
// @ts-expect-error Concrete return types do not widen to any.
const wrong: number = selectedEntity.selectors.literal();
